DELIMITER //

CREATE PROCEDURE GetAllMembers()
BEGIN
    SELECT * FROM Members;
END //

DELIMITER ;

CALL GetAllMembers();


DELIMITER //

CREATE PROCEDURE GetMemberDetails(IN p_member_id INT)
BEGIN
    SELECT *
    FROM Members
    WHERE member_id = p_member_id;
END //

DELIMITER ;

CALL GetMemberDetails(201);


DELIMITER //

CREATE PROCEDURE GetBooksWithAuthors()
BEGIN
    SELECT b.book_id, b.book_title, a.author_name, b.category, b.price
    FROM Books b
    INNER JOIN Authors a
        ON b.author_id = a.author_id;
END //

DELIMITER ;

CALL GetBooksWithAuthors();


DELIMITER //

CREATE PROCEDURE GetBorrowingStatusSummary()
BEGIN
    SELECT borrowing_status,
           COUNT(*) AS total_records
    FROM Borrowings
    GROUP BY borrowing_status;
END //

DELIMITER ;

CALL GetBorrowingStatusSummary();
